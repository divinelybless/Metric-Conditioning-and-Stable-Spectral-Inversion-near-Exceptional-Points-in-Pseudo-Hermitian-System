\section{Metric-based stability of spectral inversion}
Let $m=\min_{\lambda\in\spec(H)}\abs{\lambda}>0$.
For a perturbation $\Delta$, define
$\delta_\eta=\norm{\Delta}_\eta=\norm{\rho\Delta\rho^{-1}}_2$.
The physical metric is fixed while the perturbation is varied.

\begin{theorem}[Perturbative invertibility certificate]\label{thm:pert}
If $\delta_\eta<m$, then $H+\Delta$ is invertible and
\begin{equation}\label{eq:inverse-error}
 \norm{(H+\Delta)^{-1}-H^{-1}}_\eta
 \leq \frac{\delta_\eta}{m(m-\delta_\eta)}.
\end{equation}
In particular, the entirely Euclidean sufficient condition
\begin{equation}\label{eq:euclcertificate}
 \norm{\Delta}_2<\frac{m}{\sqrt{\kg(\eta)}}
\end{equation}
guarantees invertibility for \emph{any} perturbation $\Delta$ (which need not
preserve pseudo-Hermiticity).
\end{theorem}
\begin{proof}
The transformed perturbed operator equals
$h+E$, where $E=\rho\Delta\rho^{-1}$.
Since $\norm{h^{-1}E}_2\leq\delta_\eta/m<1$,
the Neumann series proves invertibility and
$\norm{(h+E)^{-1}}_2\leq(m-\delta_\eta)^{-1}$.
The resolvent identity and~\eqref{eq:norms} yield
\begin{equation}
 (h+E)^{-1}-h^{-1}=-(h+E)^{-1}Eh^{-1},
\end{equation}
which gives~\eqref{eq:inverse-error}. Finally,
$\delta_\eta\leq\sqrt{\kg(\eta)}\norm{\Delta}_2$.
\end{proof}

The precise Euclidean distance to a singular matrix is
\begin{equation}\label{eq:radius}
 r_{\mathrm{sing}}(H)
 :=\inf_{\Delta:\,\det(H+\Delta)=0}\norm{\Delta}_2
 =\sigmin(H)=\frac{1}{\norm{H^{-1}}_2}.
\end{equation}
Indeed, if $(H+\Delta)x=0$ for a unit $x$ then
$\norm{\Delta}\geq\norm{Hx}\geq\sigmin(H)$; equality is obtained using
$\Delta=-\sigmin(H)uv^\dagger$ for corresponding left and right
singular vectors. Combining~\eqref{eq:inversebound} and~\eqref{eq:radius}
gives the metric-based lower certificate
\begin{equation}\label{eq:radiusbound}
 r_{\mathrm{sing}}(H)\geq\frac{m}{\sqrt{\kg(\eta)}}.
\end{equation}
The lower bound is not necessarily an equality for arbitrary matrices or metrics.
